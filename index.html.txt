<script>
// آپ کا Firebase Realtime Database لنک
const FIREBASE_URL = "https://link-tracker-28893-default-rtdb.firebaseio.com";

// یہاں آپ کے لنکس ہیں (آپ مزید بھی نیچے ایڈ کر سکتے ہیں)
const links = {
"mysearch": "https://search.app/?link=https://search.shortg.xyz/r/q7zRhX9T5hVm",
"fb1": "https://your-website.com/article-1"
};

const urlParams = new URLSearchParams(window.location.search);
const code = urlParams.get('r');

// ہائی اسپیڈ ری ڈائریکشن اور کلک کاؤنٹ
if (code && links[code]) {
fetch(`${FIREBASE_URL}/clicks/${code}.json`)
.then(res => res.json())
.then(currentCount => {
const newCount = (currentCount || 0) + 1;
fetch(`${FIREBASE_URL}/clicks/${code}.json`, {
method: 'PUT',
body: JSON.stringify(newCount),
keepalive: true
});
}).catch(() => {});

window.location.replace(links[code]);
}

<script>
if (!code || !links[code]) {
document.getElementById('dashboard').style.display = 'block';
loadAnalytics();
} else {
document.getElementById('loader').style.display = 'block';
}

async function loadAnalytics() {
let clickData = {};
try {
const response = await fetch(`${FIREBASE_URL}/clicks.json`);
clickData = (await response.json()) || {};
} catch (e) {
console.error("Database connection error");
}

const labels = [];
const counts = [];
let totalAllClicks = 0;
const tableBody = document.getElementById('linksTable');
const baseUrl = window.location.origin + window.location.pathname;

for (const [key, targetUrl] of Object.entries(links)) {
const count = clickData[key] || 0;
totalAllClicks += count;
labels.push(key);
counts.push(count);

const fullShortUrl = `${baseUrl}?r=${key}`;
tableBody.innerHTML += `
<tr>
<td><strong>${key}</strong></td>
<td style="font-size: 13px; color: #94a3b8;">${fullShortUrl}</td>
<td style="color: #4ade80; font-weight: bold;">${count}</td>
<td><button class="copy-btn" onclick="copyLink('${fullShortUrl}')">Copy</button></td>
</tr>
`;
}

document.getElementById('statsCards').innerHTML = `
<div class="card"><h3>Total Active Links</h3><p>${labels.length}</p></div>
<div class="card"><h3>Total All-Time Clicks</h3><p>${totalAllClicks}</p></div>
`;

const ctx = document.getElementById('clicksChart').getContext('2d');
new Chart(ctx, {
type: 'bar',
data: {
labels: labels,
datasets: [{
label: 'Clicks per Link',
data: counts,
backgroundColor: '#38bdf8',
borderRadius: 6
}]
},
options: {
responsive: true,
scales: {
y: { beginAtZero: true, ticks: { color: '#94a3b8', stepSize: 1 }, grid: { color: '#334155' } },
x: { ticks: { color: '#94a3b8' }, grid: { display: false } }
},
plugins: { legend: { labels: { color: '#f8fafc' } } }
}
});
}

function copyLink(url) {
navigator.clipboard.writeText(url);
alert("Copied: " + url);
}
